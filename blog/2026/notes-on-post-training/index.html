Policy update

def advantage(rewards, values, gamma=0.99):
    returns = []
    g = 0.0
    for r, v in zip(reversed(rewards), reversed(values)):
        g = r + gamma * g
        returns.append(g)
    returns.reverse()
    return [g - v for g, v in zip(returns, values)]


if __name__ == "__main__":
    print(advantage([1.0, 0.0, 1.0], [0.3, 0.2, 0.4]))
